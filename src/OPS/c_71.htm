དྲེགས་པ་སྡེ་ལྔ།502

 །།ཟབ་པ་སྐོར་བདུན་ལས༔  གཏེར་སྲུང་དྲེགས་པ་སྡེ་ལྔའི་གཏོར་ཆོག་བཞུགས༔ ན་མོ་ཤྲཱི་ཧེ་རུ་ཀཱ་ཡ༔ ཟབ་བདུན་གཉེར་དུ་གཏད་པ་ཡི༔  དྲེགས་པ་སྡེ་ལྔའི་མཆོད་ཐབས་ནི༔ ཞི་བ་བྱེད་ན་དཀར་གཏོར་ལྔ༔ དྲག་པོ་བྱེད་ན་དམར་གཏོར་ལྔ༔  མཐུན་རྫས་དཀོར་ཆ་རྣམས་ཀྱིས་བྲན༔ བདག་ཉིད་ལྷར་གསལ་གཏོར་མཆོད་བརླབས༔ ཐུགས་ཀའི་འོད་ཀྱིས་དྲེགས་པ་བསྐུལ༔  ཨོཾ་ཤུ་ལ་ཏྲི་ཛ༔  ཀླུ་བཙན་གྱི་སྔགས་བཟླས་ཤིང་གཏོར་མ་ཕུལ༔  ཨོཾ་དེ་ཝ་ཨ་ཛཿ ཤར་ལྷའི་སྔགས་བཟླས་ཤིང་གཏོར་མ་འབུལ༔  ཨོཾ་དྷ་ཤ་གྷྲི་བ་དུ་ཛ༔  ལྷོ་བདུད་ཀྱི་སྔགས་བཟླས་ཤིང་གཏོར་མ་འབུལ༔  ཨོཾ་ནཱ་ག་རཱ་ཛ་ནཱཾ་ཛ༔  ནུབ་ཀླུའི་སྔགས་བཟླས་ཤིང་གཏོར་མ་འབུལ༔  ཨོཾ་དྷྭ་ཏི་ཀུ་མཱ་ར་བཻ་ཛ༔  བྱང་གནོད་སྦྱིན་གྱི་སྔགས་བཟླས་ཤིང་གཏོར་མ་འབུལ༔  པདྨ་བདག་ཉིད་ནམ་རྫོང་དུ༔ ཡིད་ཆེས་འདི་ལྔས་སྲོག་སྙིང་ཕུལ༔ གསེར་སྐྱེམས་སྦྱིན་ནས་དམ་བསྒྲགས་ཏེ༔  ཤེལ་རྔ་བརྡུངས་ཤིང་འདི་སྐད་བསྟོད༔ ཧཱུྃ༔ ཤུ་བ་སྲོག་ཟན་བཙན་གྱི་རྗེ༔ ཡམ་ཤུད་དམར་པོ་མཐུ་སྟོབས་ཅན༔  བསྭེ་ཡི་སྐྱེས་བུ་རབ་འཇིགས་པ༔ སྐུ་མདོག་དམར་རྔམ་ཁྲོས་པ་ལ༔ ཕྱག་ན་མདུང་དམར་ཞགས་པ་འཛིན༔  བསྭེ་ཁྲབ་གསོལ་ཞིང་རྟ་དམར་ཆིབས༔ དར་དམར་རྡོ་རྗེ་ཅོད་པན་ཅན༔ ཀླུ་བཙན་འབུམ་གྱི་འཁོར་གྱིས་བསྐོར༔  འདིར་གཤེགས་དམ་རྫས་གཏོར་མ་བཞེས༔ པདྨ་འབྱུང་གནས་བཀའ་བཞིན་དུ༔ སྒྲུབ་པ་པོ་ཡི་ཆོས་སྲིད་སྐྱོངས༔  ཆོས་སྐྱོང་རྒྱལ་པོའི་བསམ་པ་སྒྲུབས༔ བཅོལ་བའི་ཕྲིན་ལས་འགྲུབ་པར་མཛོད༔ ཧཱུྃ༔ ལྷ་ཡི་དགེ་བསྙེན་དྲག་རྩལ་ཅན༔  རྡོ་རྗེ་དཔལ་ལྡན་ཤེལ་གྱི་མདོག༔ རྒྱལ་མཚན་དཀར་པོ་པུསྟི་འཛིན༔ ཤེལ་ཁྲབ་གསོལ་ཞིང་རྟ་དཀར་ཆིབས༔  དར་དཀར་རྡོ་རྗེའི་ཅོད་པན་ཅན༔ ལྷ་དམག་འབུམ་གྱི་འཁོར་གྱིས་བསྐོར༔ འདིར་གཤེགས་དམ་རྫས་གཏོར་མ་བཞེས༔  སྒྲུབ་པོའི་འགལ་རྐྱེན་ཞི་བར་མཛོད༔ བཅོལ་བའི་ཕྲིན་ལས་འགྲུབ་པར་མཛོད༔ ཧཱུྃ༔ བདུད་ཀྱི་རྒྱལ་པོ་སྲོག་གི་བདག༔  བརྐོས་རྗེ་དྲང་དཀར་དམག་གི་དཔོན༔ ནག་པོ་བདུད་ཀྱི་མདུང་ཞགས་འཛིན༔ ལྕགས་ཁྲབ་གསོལ་ཞིང་བདུད་རྟ་ཆིབས༔  དར་ནག་རྡོ་རྗེའི་ཅོད་པན་ཅན༔ བདུད་དམག་འབུམ་གྱི་འཁོར་གྱིས་བསྐོར༔ འདིར་གཤེགས་དམ་རྫས་གཏོར་མ་བཞེས༔  སྒྲུབ་པ་པོ་ལ་སྡང་བྱེད་སྒྲོལ༔ བཅོལ་བའི་ཕྲིན་ལས་འགྲུབ་པར་མཛོད༔ ཧཱུྃ༔ ཀླུ་བདུད་ནག་པོ་སོག་མ་མེད༔  ཀླུ་ཡི་རྒྱལ་པོ་དམར་སེར་མདོག༔ ཕྱག་ན་མདུང་དར་སྦྲུལ་ཞགས་འཛིན༔ གཡུ་ཁྲབ་གསོལ་ཞིང་ཀླུ་རྟ་ཆིབས༔  དར་དམར་རྡོ་རྗེའི་ཅོད་པན་ཅན༔ ཀླུ་དང་ས་བདག་འབུམ་གྱིས་བསྐོར༔ འདིར་གཤེགས་དམ་རྫས་གཏོར་མ་བཞེས༔  སྒྲུབ་པ་པོ་ཡི་སྙན་པ་སྒྲོགས༔ བཅོལ་བའི་ཕྲིན་ལས་འགྲུབ་པར་མཛོད༔ ཧཱུྃ༔ རྡོ་རྗེའི་གནོད་སྦྱིན་ནོར་བདག་ནི༔  སྐུ་མདོག་ངང་སེར་འོད་དུ་འབར༔ ཕྱག་ན་རྒྱལ་མཚན་ནེའུ་ལེ་འཛིན༔ བསེ་ཁྲབ་གསོལ་ཞིང་གཉན་རྟ་ཆིབས༔  དར་སེར་རྡོ་རྗེའི་ཅོད་པན་ཅན༔ གཉན་ཆེན་འབུམ་གྱི་འཁོར་གྱིས་བསྐོར༔ འདིར་གཤེགས་དམ་རྫས་གཏོར་མ་བཞེས༔  སྒྲུབ་པ་པོ་ཡི་ཚེ་གཡང་སྤེལ༔ ནོར་དང་ལོངས་སྤྱོད་དངོས་གྲུབ་སྩོལ༔ བཅོལ་བའི་ཕྲིན་ལས་འགྲུབ་པར་མཛོད༔  གཏོར་མ་གཙང་སར་འབུལ་ཞིང་ཕྲིན་ལས་བཅོལ༔  རང་རང་གི་གནས་སུ་གཏང་བར་བྱའོ༔ ས་མ་ཡ༔ རྒྱ་རྒྱ་རྒྱ༔ ཁོ་བོ་མཆོག་གྱུར་བདེ་ཆེན་གླིང་པས་རང་ལོ་ཉེར་བརྒྱད་སྒྲ་རྒྱལ་འབྲུག་གི་ལོ་ལ་ཀརྨའི་དམ་ཅན་བྲག་ནས་ཕུར་སྒྲོམ་གདན་དྲངས་པའི་ཟབ་པ་སྐོར་བདུན་ལས་ཕྱིའི་གཏེར་སྲུང་མཆོད་ཐབས་འདི་བཙན་རྒོད་ཆེན་པོའི་རི་རྩེ་མདོ་ཁམས་ཀུན་གྱི་མགུལ་ལྷ་ཐམས་ཅད་མཐོང་སར་ཕབ་པའི་ཡི་གེ་པ་ནི་ཟུར་མང་རྟོགས་ལྡན་དཀོན་མཆོག་རྒྱལ་མཚན་གྱིས་བགྱིས་ཤིང་། དེའི་ཉིན་ཤེལ་གྱི་ཁ་དོག་དང་འདྲ་བའི་ཆར་དང་། འཇའ་ཡི་ཀ་བ་བཞི་བཙུགས་པ་སོགས་ལྟས་དགེ་བར་བྱུང་ཞིང་། རང་ཉམས་ལའང་བོད་ཁམས་ཀྱི་བསྟན་འགྲོར་ཕན་ནམ་སྙམ་པ་དང་། ལྷག་པར་ནུབ་ཀླུ་བདུད་དགེ་བསྙེན་ན་བུན་རྫོང་གི་རང་གཟུགས་བསྟན་ནས་བསུ་བར་བྱུང་། བྱང་གཉན་དྷྭ་ཏིས་རང་གཟུགས་བསྟན་ཏེ་ཤར་ཕྱོགས་སུ་ཕྲིན་ལས་སྒྲུབ་པར་སོང་བ་དང་གཞན་ཡང་ཡུལ་ལྷ་རྣམས་ཀྱིས་ལྟས་ངོ་མཚར་བ་རེ་བསྟན་བྱུང་བ་བཅས། ཨོ་རྒྱན་ཆེན་པོའི་ཐུགས་རྗེ་ཚད་མེད་པས་བོད་ཁམས་ཀུན་ལ་རྣམ་པ་ཀུན་ཏུ་དགེ་བར་གྱུར་ཅིག།
